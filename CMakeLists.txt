cmake_minimum_required(VERSION 3.20)
project(InputOverlay VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "Input Overlay requires Windows 10 or newer.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
find_package(Git QUIET)
set(INPUT_OVERLAY_BUILD_COMMIT "")
if(GIT_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE INPUT_OVERLAY_BUILD_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
if(NOT INPUT_OVERLAY_BUILD_COMMIT MATCHES "^[0-9a-f]+$" OR NOT INPUT_OVERLAY_BUILD_COMMIT MATCHES "^........................................$")
    set(INPUT_OVERLAY_BUILD_COMMIT "")
endif()
add_compile_definitions(INPUT_OVERLAY_BUILD_COMMIT="${INPUT_OVERLAY_BUILD_COMMIT}" INPUT_OVERLAY_VERSION="${PROJECT_VERSION}")
option(INPUT_OVERLAY_BUILD_MANUAL_TESTS "Build the interactive fullscreen and click-through test window" OFF)
if(INPUT_OVERLAY_BUILD_MANUAL_TESTS)
    add_executable(window_fixture WIN32 tests/window_fixture.cpp)
    target_link_libraries(window_fixture PRIVATE user32 gdi32)
    if(NOT MSVC)
        target_link_options(window_fixture PRIVATE -municode)
    endif()
endif()
add_executable(InputOverlay WIN32 src/main.cpp src/config.cpp src/settings.cpp src/overlay.cpp src/injected_input.cpp src/controller.cpp src/updates.cpp src/update_install.cpp src/app.rc)
set_target_properties(InputOverlay PROPERTIES OUTPUT_NAME "Input Overlay")
target_include_directories(InputOverlay PRIVATE src)
target_link_libraries(InputOverlay PRIVATE user32 gdi32 gdiplus comctl32 shell32 advapi32 dwmapi wtsapi32 uxtheme winhttp bcrypt)
if(MSVC)
    target_compile_options(InputOverlay PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>")
    target_compile_options(InputOverlay PRIVATE "$<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:/Brepro>")
    target_link_options(InputOverlay PRIVATE /MANIFEST:NO "$<$<CONFIG:Release>:/DEBUG:NONE;/INCREMENTAL:NO;/Brepro>")
else()
    target_compile_options(InputOverlay PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>")
    target_compile_options(InputOverlay PRIVATE "$<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:-g0;-fno-ident;-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.;-ffile-prefix-map=${CMAKE_BINARY_DIR}=./build>")
    target_link_options(InputOverlay PRIVATE -municode -static "$<$<CONFIG:Release>:-s;-Wl,--build-id=none>")
    include(CheckLinkerFlag)
    check_linker_flag(CXX "-Wl,--no-insert-timestamp" INPUT_OVERLAY_HAS_NO_INSERT_TIMESTAMP)
    if(INPUT_OVERLAY_HAS_NO_INSERT_TIMESTAMP)
        target_link_options(InputOverlay PRIVATE "$<$<CONFIG:Release>:-Wl,--no-insert-timestamp>")
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp src/config.cpp)
    target_include_directories(core_tests PRIVATE src)
    target_link_libraries(core_tests PRIVATE user32 advapi32 dwmapi)
    if(MSVC)
        target_compile_options(core_tests PRIVATE /utf-8)
    endif()
    add_test(NAME core COMMAND core_tests)
    add_executable(overlay_smoke tests/overlay_smoke.cpp src/overlay.cpp)
    target_include_directories(overlay_smoke PRIVATE src)
    target_link_libraries(overlay_smoke PRIVATE user32 gdi32 gdiplus)
    if(MSVC)
        target_compile_options(overlay_smoke PRIVATE /utf-8)
    endif()
    add_test(NAME overlay COMMAND overlay_smoke)
    add_executable(injected_input_tests tests/injected_input_tests.cpp src/injected_input.cpp)
    target_include_directories(injected_input_tests PRIVATE src)
    target_link_libraries(injected_input_tests PRIVATE user32)
    add_test(NAME input_worker COMMAND injected_input_tests)
    add_executable(controller_tests tests/controller_tests.cpp src/controller.cpp)
    target_include_directories(controller_tests PRIVATE src)
    add_test(NAME controller COMMAND controller_tests)
    add_executable(update_tests tests/update_tests.cpp src/updates.cpp)
    target_include_directories(update_tests PRIVATE src)
    target_link_libraries(update_tests PRIVATE winhttp bcrypt)
    add_test(NAME updates COMMAND update_tests)
    add_executable(update_install_tests tests/update_install_tests.cpp src/update_install.cpp)
    target_include_directories(update_install_tests PRIVATE src)
    target_link_libraries(update_install_tests PRIVATE bcrypt shell32)
    add_test(NAME update_install COMMAND update_install_tests)
    if(MSVC)
        target_compile_options(update_tests PRIVATE /utf-8)
        target_compile_options(update_install_tests PRIVATE /utf-8)
    endif()
endif()
